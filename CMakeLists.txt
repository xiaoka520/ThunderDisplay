cmake_minimum_required(VERSION 3.21)
project(ThunderDisplay LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(MINGW)
    add_link_options(-static)
endif()
enable_testing()
find_package(Threads REQUIRED)
add_executable(latest-picture-tests tests/latest_picture_tests.cpp)
target_include_directories(latest-picture-tests PRIVATE shared)
target_link_libraries(latest-picture-tests PRIVATE Threads::Threads)
add_test(NAME latest-picture COMMAND latest-picture-tests)
add_executable(video-inbox-tests tests/video_inbox_tests.cpp)
target_include_directories(video-inbox-tests PRIVATE shared)
target_link_libraries(video-inbox-tests PRIVATE Threads::Threads)
add_test(NAME video-inbox COMMAND video-inbox-tests)
add_executable(recovery-tests tests/recovery_tests.cpp)
target_include_directories(recovery-tests PRIVATE shared)
add_test(NAME recovery COMMAND recovery-tests)
add_executable(blob-tests tests/blob_tests.cpp)
target_include_directories(blob-tests PRIVATE shared)
add_test(NAME blob COMMAND blob-tests)
add_executable(protocol-tests tests/protocol_tests.cpp)
target_include_directories(protocol-tests PRIVATE shared)
add_test(NAME protocol COMMAND protocol-tests)
add_executable(setup-layout-tests tests/setup_layout_tests.cpp)
target_include_directories(setup-layout-tests PRIVATE windows-client/src)
add_test(NAME setup-layout COMMAND setup-layout-tests)
add_executable(quality-tests tests/quality_tests.cpp)
target_include_directories(quality-tests PRIVATE shared)
add_test(NAME quality COMMAND quality-tests)
add_executable(clipboard-tests tests/clipboard_tests.cpp)
target_include_directories(clipboard-tests PRIVATE shared)
add_test(NAME clipboard COMMAND clipboard-tests)
add_executable(shortcut-tests tests/shortcut_tests.cpp)
target_include_directories(shortcut-tests PRIVATE windows-client/src)
add_test(NAME shortcut COMMAND shortcut-tests)
add_executable(presentation-tests tests/presentation_tests.cpp)
target_include_directories(presentation-tests PRIVATE shared)
add_test(NAME presentation COMMAND presentation-tests)
if(WIN32)
    add_subdirectory(windows-client)
endif()
